    36→/* ---------- helpers ---------- */
    37→function scopeCss(css: string, scope: string): string {
    38→  const noComments = css.replace(/\/\*[\s\S]*?\*\//g, "");
    39→  return noComments.replace(/([^{}]+)\{([^{}]*)\}/g, (match: string, sel: string, body: string) => {
    40→    const trimmed = sel.trim();
    41→    if (!trimmed || trimmed.startsWith("@")) return match;
    42→    const scoped = trimmed
    43→      .split(",")
    44→      .map((s) => `.${scope} ${s.trim()}`)
    45→      .join(", ");
    46→    return `${scoped} {${body}}`;
    47→  });
    48→}
    49→
    50→function patternMatches(url: string, pattern: string): boolean {
    51→  try {
    52→    const regexStr = pattern.replace(/[.+?^${}()|[\]\\]/g, "\\$&").replace(/\*/g, ".*");
    53→    return new RegExp(regexStr, "i").test(url);
    54→  } catch {
    55→    return false;
    56→  }
    57→}
