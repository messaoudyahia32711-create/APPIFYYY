    55→def esc_xml(s: str) -> str:
    56→    return (s or "").replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;").replace('"', "&quot;").replace("'", "&apos;")

    59→def esc_js(s: str) -> str:
    60→    return (s or "").replace("\\", "\\\\").replace('"', '\\"').replace("\n", "\\n").replace("\r", "")

    63→def sanitize_package(p: str) -> str:
    64→    p = (p or "").strip().lower()
    65→    p = re.sub(r"[^a-z0-9_.]", "", p)
    66→    segs = [s for s in p.split(".") if s]
    67→    segs = [s if s[0].isalpha() else ("a" + s) for s in segs]
    68→    while len(segs) < 2:
    69→        segs.append("app")
    70→    return ".".join(segs[:6])
    71→
    72→
    73→def sanitize_dir(s: str) -> str:
    74→    s = re.sub(r"[\\/:*?\"<>|\x00-\x1f]", "-", (s or "app")).strip(". ")
    75→    return (s or "app")[:64]

    78→def parse_color(c: str, fallback: str) -> str:
    79→    c = (c or "").strip()
    80→    if re.fullmatch(r"#[0-9a-fA-F]{6}", c):
    81→        return c
    82→    if re.fullmatch(r"#[0-9a-fA-F]{8}", c):
    83→        return "#" + c[3:]
    84→    return fallback

    88→def make_icons(res_dir: str, icon_dataurl, name: str):
    89→    """Generate legacy launcher icons + adaptive (API26+) foreground icons."""
    90→    icon_img = None
    91→    if icon_dataurl and isinstance(icon_dataurl, str) and icon_dataurl.startswith("data:"):
    92→        try:
    93→            import base64
    94→            b64 = icon_dataurl.split(",", 1)[1]
    95→            raw = base64.b64decode(b64)
    96→            img = Image.open(BytesIO(raw)).convert("RGBA")
    97→            icon_img = img.resize((512, 512), Image.LANCZOS)
    98→        except Exception:
    99→            icon_img = None
   100→
   101→    if icon_img is None:
   102→        # default: purple gradient rounded square with a white bolt glyph
   103→        img = Image.new("RGBA", (512, 512), (0, 0, 0, 0))
   104→        top, bottom = (109, 40, 217), (139, 92, 246)
   105→        for y in range(512):
   106→            t = y / 511.0
   107→            row = tuple(int(top[i] + (bottom[i] - top[i]) * t) for i in range(3))
   108→            ImageDraw.Draw(img).rectangle([0, y, 512, y], fill=row + (255,))
   109→        mask = Image.new("L", (512, 512), 0)
   110→        ImageDraw.Draw(mask).rounded_rectangle([0, 0, 511, 511], radius=96, fill=255)
   111→        icon_img = Image.new("RGBA", (512, 512), (0, 0, 0, 0))
   112→        icon_img.paste(img, (0, 0), mask)
   113→        d = ImageDraw.Draw(icon_img)
   114→        # lightning bolt
   115→        bolt = [(300, 92), (188, 280), (258, 280), (212, 420), (330, 238), (256, 238)]
   116→        d.polygon(bolt, fill=(255, 255, 255, 255))
   117→
   118→    densities = {"mdpi": 48, "hdpi": 72, "xhdpi": 96, "xxhdpi": 144, "xxxhdpi": 192}
   119→    fg = {"mdpi": 108, "hdpi": 162, "xhdpi": 216, "xxhdpi": 324, "xxxhdpi": 432}
   120→
   121→    for dpi, size in densities.items():
   122→        d = os.path.join(res_dir, "mipmap-" + dpi)
   123→        os.makedirs(d, exist_ok=True)
   124→        icon_img.resize((size, size), Image.LANCZOS).save(os.path.join(d, "ic_launcher.png"))
   125→
   126→    for dpi, canvas in fg.items():
   127→        d = os.path.join(res_dir, "mipmap-" + dpi)
   128→        content = int(canvas * 0.62)
   129→        icon = icon_img.resize((content, content), Image.LANCZOS)
   130→        layer = Image.new("RGBA", (canvas, canvas), (0, 0, 0, 0))
   131→        off = (canvas - content) // 2
   132→        layer.paste(icon, (off, off), icon)
   133→        layer.save(os.path.join(d, "ic_launcher_fg.png"))
   134→
   135→    ad = os.path.join(res_dir, "mipmap-anydpi-v26")
   136→    os.makedirs(ad, exist_ok=True)
   137→    with open(os.path.join(ad, "ic_launcher.xml"), "w", encoding="utf-8") as f:
   138→        f.write(
   139→            '<?xml version="1.0" encoding="utf-8"?>\n'
   140→            '<adaptive-icon xmlns:android="http://schemas.android.com/apk/res/android">\n'
   141→            '    <background android:drawable="@color/adaptiveBg"/>\n'
   142→            '    <foreground android:drawable="@mipmap/ic_launcher_fg"/>\n'

   194→# ------------------------------------------------------------- manifest ---
   195→def write_manifest(build_dir: str, cfg: dict, pkg: str, name: str, ver_name: str, ver_code: int):
   196→    st = cfg.get("settings") or {}
   197→    style = cfg.get("style") or {}
   198→    deeplink = cfg.get("deepLink") or {}
   199→    mon = cfg.get("monetization") or {}
   200→
   201→    perms = [
   202→        '    <uses-permission android:name="android.permission.INTERNET"/>',
   203→        '    <uses-permission android:name="android.permission.ACCESS_NETWORK_STATE"/>',
   204→        '    <uses-permission android:name="android.permission.USE_BIOMETRIC"/>',
   205→    ]
   206→    if st.get("camera") or st.get("qrScanner"):
   207→        perms.append('    <uses-permission android:name="android.permission.CAMERA"/>')
   208→        perms.append('    <uses-feature android:name="android.hardware.camera" android:required="false"/>')
   209→    if st.get("microphone"):
   210→        perms.append('    <uses-permission android:name="android.permission.RECORD_AUDIO"/>')
   211→        perms.append('    <uses-permission android:name="android.permission.MODIFY_AUDIO_SETTINGS"/>')
   212→    if st.get("location"):
   213→        perms.append('    <uses-permission android:name="android.permission.ACCESS_FINE_LOCATION"/>')
   214→        perms.append('    <uses-permission android:name="android.permission.ACCESS_COARSE_LOCATION"/>')
   215→    if st.get("vibration"):
   216→        perms.append('    <uses-permission android:name="android.permission.VIBRATE"/>')
   217→    if cfg.get("hasNotifications"):
   218→        perms.append('    <uses-permission android:name="android.permission.POST_NOTIFICATIONS"/>')
   219→
   220→    orientation = {"landscape": "landscape", "auto": "fullSensor"}.get(style.get("orientation"), "portrait")
   221→
   222→    meta = ""
   223→    if mon.get("admobEnabled") and (mon.get("admobAppIdAndroid") or "").strip():
   224→        meta = (
   225→            '        <meta-data android:name="com.google.android.gms.ads.APPLICATION_ID"\n'
   226→            f'            android:value="{esc_xml(mon["admobAppIdAndroid"])}"/>\n'
   227→        )
   228→
   229→    deeplink_xml = ""
   230→    scheme = (deeplink.get("scheme") or "").strip()
   231→    host = (deeplink.get("host") or "").strip()
   232→    if scheme:
   233→        host_attr = f' android:host="{esc_xml(host)}"' if host else ""
   234→        deeplink_xml = (
   235→            '            <intent-filter android:autoVerify="false">\n'
   236→            '                <action android:name="android.intent.action.VIEW"/>\n'
   237→            '                <category android:name="android.intent.category.DEFAULT"/>\n'
   238→            '                <category android:name="android.intent.category.BROWSABLE"/>\n'
   239→            f'                <data android:scheme="{esc_xml(scheme)}"{host_attr}/>\n'
   240→            "            </intent-filter>\n"
   241→        )
   242→
   243→    cleartext = "true" if (cfg.get("startUrl") or "").startswith("http://") else "false"
   244→
   245→    manifest = (
   246→        '<?xml version="1.0" encoding="utf-8"?>\n'
   247→        '<manifest xmlns:android="http://schemas.android.com/apk/res/android"\n'
   248→        f'    package="{pkg}"\n'
   249→        f'    android:versionCode="{ver_code}"\n'
   250→        f'    android:versionName="{esc_xml(ver_name)}">\n\n'
   251→        + "\n".join(perms)
   252→        + f'\n\n    <application\n        android:label="@string/app_name"\n'
   253→        '        android:icon="@mipmap/ic_launcher"\n'
   254→        '        android:theme="@style/AppifyTheme"\n'
   255→        f'        android:usesCleartextTraffic="{cleartext}"\n'
   256→        '        android:hardwareAccelerated="true"\n'
   257→        '        android:supportsRtl="true">\n'
   258→        + meta
   259→        + '        <activity\n            android:name=".MainActivity"\n'
   260→        '            android:exported="true"\n'
   261→        f'            android:screenOrientation="{orientation}"\n'
   262→        '            android:configChanges="orientation|screenSize|keyboardHidden|screenLayout|smallestScreenSize|uiMode"\n'
   263→        '            android:windowSoftInputMode="adjustResize">\n'
   264→        '            <intent-filter>\n'
   265→        '                <action android:name="android.intent.action.MAIN"/>\n'
   266→        '                <category android:name="android.intent.category.LAUNCHER"/>\n'
   267→        "            </intent-filter>\n"
   268→        + deeplink_xml
   269→        + "        </activity>\n"
   270→        '        <service android:name=".AppifyMessagingService" android:exported="false"/>\n'
   271→        "    </application>\n</manifest>\n"
   272→    )
   273→    with open(os.path.join(build_dir, "AndroidManifest.xml"), "w", encoding="utf-8") as f:
   274→        f.write(manifest)

   765→    /* --------------------------------------------------------- routing */
   766→    boolean handleNavigation(String url) {
   767→        Uri u = Uri.parse(url);
   768→        String scheme = u.getScheme() == null ? "" : u.getScheme().toLowerCase();
   769→        if (scheme.equals("http") || scheme.equals("https")) {
   770→            JSONArray pats = settingsCfg.optJSONArray("externalUrlPatterns");
   771→            if (pats != null) {
   772→                for (int i = 0; i < pats.length(); i++) {
   773→                    if (wildcardMatch(pats.optString(i), url)) { openExternal(url); return true; }
   774→                }
   775→            }
   776→            return false;
   777→        }
   778→        if (scheme.equals(deeplinkScheme.toLowerCase())) { handleDeepLink(url); return true; }
   779→        if (scheme.equals("mailto") || scheme.equals("tel") || scheme.equals("sms")
   780→                || scheme.equals("whatsapp") || scheme.equals("intent")) {
   781→            openExternal(url);
   782→            return true;
   783→        }
   784→        return false;
   785→    }
   786→
   787→    boolean wildcardMatch(String pattern, String url) {
   788→        if (pattern == null || pattern.length() == 0) return false;
   789→        String rx = pattern.replace("\\", "\\\\").replace(".", "\\.").replace("*", ".*");
   790→        try { return url.toLowerCase().matches("(?i).*" + rx + ".*"); }
   791→        catch (Exception e) { return false; }
   792→    }

   799→        }
   800→    }

   802→    void handleDeepLink(String input) {
   803→        try {
   804→            String target = input;
   805→            if (!input.contains("://")) {
   806→                String host = "";
   807→                JSONObject dl = cfg.optJSONObject("deepLink");
   808→                if (dl != null) host = dl.optString("host", "");
   809→                target = deeplinkScheme + "://" + host + (input.startsWith("/") ? input : "/" + input);
   810→            }
   811→            startActivity(new Intent(Intent.ACTION_VIEW, Uri.parse(target)));
   812→        } catch (Exception e) {
   813→            toast("رابط غير صالح");
   814→        }
   815→    }
